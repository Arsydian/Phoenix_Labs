CREATE TABLE modules (
    id TEXT PRIMARY KEY,
    track_id TEXT NOT NULL,
    slug TEXT UNIQUE NOT NULL,
    title TEXT NOT NULL,
    difficulty TEXT CHECK(difficulty IN ('Fundamentals', 'Intermediate', 'Advanced')) NOT NULL,
    order_index INTEGER NOT NULL,
    estimated_minutes INTEGER NOT NULL,

-- 3. Zero Trust Authenticated Student Profiles

    id TEXT PRIMARY KEY,                -- Email from Cf-Access-Authenticated-User-Email

    display_name TEXT NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

);

import { cors } from 'hono/cors';

type Bindings = {
  DB: D1Database;
  ASSETS: Fetcher;
  CF_ACCESS_TEAM_DOMAIN?: string;

type Variables = {
  studentEmail: string;
};

const app = new Hono<{ Bindings: Bindings; Variables: Variables }>();

app.use('/api/*', cors());

/**
 * CLOUDFLARE ZERO TRUST ACCESS MIDDLEWARE
 * Automatically extracts the authenticated user identity injected by Cloudflare Access
 * (Cf-Access-Authenticated-User-Email) and ensures their student record exists in D1.
 * Falls back to 'local-operator@localhost' when running locally via 'wrangler dev'.
 */
app.use('/api/*', async (c, next) => {
  const cfEmail =
    c.req.header('Cf-Access-Authenticated-User-Email') || 'local-operator@localhost';
  const displayName = cfEmail.split('@')[0];

  await c.env.DB.prepare(
    \`INSERT INTO students (id, email, display_name, last_active_at)
     VALUES (?, ?, ?, CURRENT_TIMESTAMP)
     ON CONFLICT(id) DO UPDATE SET last_active_at = CURRENT_TIMESTAMP\`
  )
    .bind(cfEmail, cfEmail, displayName)
    .run();

  c.set('studentEmail', cfEmail);
  await next();
});

// 1. Get Student Identity, Tracks, Modules, and Per-Track Completion Rollup
app.get('/api/curriculum', async (c) => {
  const studentId = c.get('studentEmail');

  const { results: tracks } = await c.env.DB.prepare(
    \`SELECT * FROM tracks ORDER BY order_index ASC\`
  ).all();

  const { results: modules } = await c.env.DB.prepare(
    \`SELECT m.id, m.track_id, m.slug, m.title, m.difficulty, m.order_index,
            m.estimated_minutes, m.lab_type,
            COALESCE(p.status, 'not_started') as status,
            COALESCE(p.attempts, 0) as attempts,
            COALESCE(p.time_spent_seconds, 0) as time_spent_seconds,
            p.completed_at
     FROM modules m
     LEFT JOIN student_module_progress p
       ON m.id = p.module_id AND p.student_id = ?
     ORDER BY m.order_index ASC\`
  )
    .bind(studentId)
    .all();

  return c.json({
    student: {
      email: studentId,
      zero_trust_Enforced: Boolean(c.req.header('Cf-Access-Authenticated-User-Email'))
    },
    tracks,
    modules
  });
});

// 2. Load Module Content & Automatically Log 'in_progress' Start Checkpoint
app.get('/api/modules/:slug', async (c) => {
  const studentId = c.get('studentEmail');
  const slug = c.req.param('slug');

  const mod = await c.env.DB.prepare(
    \`SELECT id, track_id, slug, title, difficulty, estimated_minutes,
            theory_json, lab_type, lab_briefing, lab_config_json, hint
     FROM modules WHERE slug = ?\`
  )
    .bind(slug)
    .first();

  if (!mod) return c.json({ error: 'Module not found' }, 404);

  // Record that the student has started or resumed this module
  await c.env.DB.prepare(
    \`INSERT INTO student_module_progress (student_id, module_id, track_id, status, started_at, last_accessed_at)
     VALUES (?, ?, ?, 'in_progress', CURRENT_TIMESTAMP, CURRENT_TIMESTAMP)
     ON CONFLICT(student_id, module_id) DO UPDATE SET
       last_accessed_at = CURRENT_TIMESTAMP\`
  )
    .bind(studentId, mod.id, mod.track_id)
    .run();

  const progress = await c.env.DB.prepare(
    \`SELECT status, attempts, hints_used, saved_lab_state_json, time_spent_seconds, completed_at
     FROM student_module_progress
     WHERE student_id = ? AND module_id = ?\`
  )
    .bind(studentId, mod.id)
    .first();

  return c.json({
    ...mod,
    theory_sections: JSON.parse(mod.theory_json),
    lab_config: JSON.parse(mod.lab_config_json),
    progress: {
      ...progress,
      saved_lab_state: progress?.saved_lab_state_json
        ? JSON.parse(progress.saved_lab_state_json)
        : {}
    }
  });
});

// 3. Heartbeat & Lab State Autosave (Saves terminal history, time spent, and hints used)
app.patch('/api/modules/:id/autosave', async (c) => {
  const studentId = c.get('studentEmail');
  const moduleId = c.req.param('id');
  const { elapsed_seconds, hints_used, saved_lab_state } = await c.req.json();

  await c.env.DB.prepare(
    \`UPDATE student_module_progress
     SET time_spent_seconds = time_spent_seconds + ?,
         hints_used = COALESCE(?, hints_used),
         saved_lab_state_json = COALESCE(?, saved_lab_state_json),
         last_accessed_at = CURRENT_TIMESTAMP
     WHERE student_id = ? AND module_id = ?\`
  )
    .bind(
      elapsed_seconds || 0,
      hints_used ?? null,
      saved_lab_state ? JSON.stringify(saved_lab_state) : null,
      studentId,
      moduleId
    )
    .run();

  return c.json({ ok: true });
});

// 4. Verify Lab Submission / Flag & Record Completion in D1
app.post('/api/modules/:id/verify', async (c) => {
  const studentId = c.get('studentEmail');
  const moduleId = c.req.param('id');
  const { submission } = await c.req.json<{ submission: string }>();

  const mod = await c.env.DB.prepare(
    \`SELECT id, track_id, expected_flag FROM modules WHERE id = ?\`
  )
    .bind(moduleId)
    .first<{ id: string; track_id: string; expected_flag: string }>();

  if (!mod) return c.json({ error: 'Module not found' }, 404);

  const isCorrect =
    submission.trim().toLowerCase() === mod.expected_flag.trim().toLowerCase();

  await c.env.DB.prepare(
    \`INSERT INTO student_module_progress (student_id, module_id, track_id, status, attempts, completed_at, last_accessed_at)
     VALUES (?, ?, ?, ?, 1, CASE WHEN ? THEN CURRENT_TIMESTAMP ELSE NULL END, CURRENT_TIMESTAMP)
     ON CONFLICT(student_id, module_id) DO UPDATE SET
       attempts = student_module_progress.attempts + 1,
       status = CASE WHEN ? THEN 'completed' ELSE student_module_progress.status END,
       completed_at = CASE WHEN ? THEN CURRENT_TIMESTAMP ELSE student_module_progress.completed_at END,
       last_accessed_at = CURRENT_TIMESTAMP\`
  )
    .bind(
      studentId,
      moduleId,
      mod.track_id,
      isCorrect ? 'completed' : 'in_progress',
      isCorrect ? 1 : 0,
      isCorrect ? 1 : 0,
      isCorrect ? 1 : 0
    )
    .run();

  return c.json({
    correct: isCorrect,
    message: isCorrect
      ? 'Verification successful. Progress committed to Cloudflare D1.'
      : 'Verification mismatch. Review the telemetry output or check the module hint.'
  });
});

    )

    let currentFileKey = Object.keys(PROJECT_FILES)[0];

    function renderFileList() {
      const container = document.getElementById('fileList');
      container.innerHTML = '';
      Object.keys(PROJECT_FILES).forEach((filePath) => {
        const isSelected = filePath === currentFileKey;
        const btn = document.createElement('button');
        btn.className = `w-full text-left px-3.5 py-2.5 rounded-lg border font-mono text-xs transition flex items-center justify-between ${
          isSelected
            ? 'bg-slate-800 border-emerald-500/50 text-emerald-300 font-semibold'
            : 'bg-slate-900/60 border-slate-800 text-slate-300 hover:bg-slate-900'
        }`;
        btn.onclick = () => selectFile(filePath);
        btn.innerHTML = `${filePath}View`;
        container.appendChild(btn);
      });
    }

    function selectFile(filePath) {
      currentFileKey = filePath;
      document.getElementById('activeFileName').textContent = filePath;
      document.getElementById('activeFileContent').textContent = PROJECT_FILES[filePath];
      renderFileList();
    }

    function downloadActiveFile() {
      const content = PROJECT_FILES[currentFileKey];
      const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = currentFileKey.split('/').pop();
      a.click();
      URL.revokeObjectURL(url);
    }

    function copyActiveFile() {
      navigator.clipboard.writeText(PROJECT_FILES[currentFileKey]);
      const btn = document.getElementById('copyBtn');
      btn.textContent = '✓ Copied';
      setTimeout(() => (btn.textContent = 'Copy Code'), 1500);
    }

    async function downloadFullZip() {
      const btn = document.getElementById('downloadZipBtn');
      btn.textContent = '⏳ Packaging ZIP...';
      const zip = new JSZip();
      Object.entries(PROJECT_FILES).forEach(([path, content]) => {
        zip.file(path, content);
      });
      const blob = await zip.generateAsync({ type: 'blob' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'secops-academy-cloudflare.zip';
      a.click();
      URL.revokeObjectURL(url);
      btn.textContent = '✓ Downloaded (.ZIP)';
      setTimeout(() => (btn.textContent = '⬇ Download Full Project (.ZIP)'), 2500);
    }

    // Initialize viewer
    selectFile(currentFileKey);
